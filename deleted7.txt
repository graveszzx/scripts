local function getRemote()
    local bridge = RS:FindFirstChild("ffrostflame_bridgenet2@1.0.0")
    return bridge and bridge:FindFirstChild("dataRemoteEvent")
end

local function getHRP()
    local c = lp.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function teleport(pos)
    local hrp = getHRP()
    if hrp then hrp.CFrame = CFrame.new(pos) end
end

local function fireRemote(args)
    local remote = getRemote()
    if not remote then return end
    pcall(function() remote:FireServer(unpack(args)) end)
end

local function isGamemodeGuiActive()
    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return false end
    local gm = pg:FindFirstChild("Gamemode")
    return gm and gm.Enabled
end

local function getGamemodeLabel()
    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return nil end
    local gm = pg:FindFirstChild("Gamemode")
    if not gm or not gm.Enabled then return nil end
    local lbl = gm:FindFirstChild("Content")
        and gm.Content:FindFirstChild("Start")
        and gm.Content.Start:FindFirstChild("Label")
    return lbl
end

local function getCurrentWaveOrStage()
    local lbl = getGamemodeLabel()
    if not lbl then return nil, nil end
    local ctx = (lbl.ContentText ~= "" and lbl.ContentText)
        or (lbl.Text ~= "" and lbl.Text)
        or ""
    local wave = ctx:match("Wave:%s*(%d+)")
    if wave then return tonumber(wave), "Wave" end
    local stage = ctx:match("Stage:%s*(%d+)")
    if stage then return tonumber(stage), "Stage" end
    return nil, nil
end

local function isGuiShowingWave()
    local lbl = getGamemodeLabel()
    if not lbl then return false end
    local ctx = (lbl.ContentText ~= "" and lbl.ContentText)
        or (lbl.Text ~= "" and lbl.Text)
        or ""
    return ctx:match("Wave:") ~= nil
end

local function isGuiShowingStage()
    local lbl = getGamemodeLabel()

    local ctx = (lbl.ContentText ~= "" and lbl.ContentText)
        or (lbl.Text ~= "" and lbl.Text)
        or ""
    return ctx:match("Stage:") ~= nil
end

local function hasLiveEnemiesInRange()
    local inRange = workspace:FindFirstChild("_ENEMIES")
        and workspace._ENEMIES:FindFirstChild("InRange")
    if not inRange then return false end
    for _, child in ipairs(inRange:GetChildren()) do
        if child.Name ~= "Highlight" then return true end
    end
    return false
end

local function getInRangeIDs()
    local inRange = workspace:FindFirstChild("_ENEMIES")
        and workspace._ENEMIES:FindFirstChild("InRange")
    if not inRange then return {} end
    local ids = {}
    for _, child in ipairs(inRange:GetChildren()) do
        if child.Name ~= "Highlight" then
            table.insert(ids, child.Name)
        end
    end
    return ids
end

local function labelOrb(root)
    if root:FindFirstChild("_collected") then return end
    local tag = Instance.new("StringValue")
    tag.Name = "_collected"
    tag.Parent = root
end

local function isOrbCollected(root)
    return root:FindFirstChild("_collected") ~= nil
end

local function getUncollectedOrbs()
    local spawnItems = workspace:FindFirstChild("_SPAWNITEMS")
    if not spawnItems then return {} end
    local orbs = {}
    for _, item in ipairs(spawnItems:GetChildren()) do
        local root = item:FindFirstChild("Root")
        local pp   = root and root:FindFirstChild("ProximityPrompt")
        if root and pp and not isOrbCollected(root) then
            table.insert(orbs, { root = root, prompt = pp })
        end
    end
    return orbs
end

local function collectAllOrbs()
    if State.collectingOrbs then return end
    State.collectingOrbs = true
    for i = 1, 8 do teleport(FOX_SIN_POS); task.wait(0.3) end
    task.wait(0.5)
    for _, spotPos in ipairs(ORB_SPOTS) do
        if not State.autoCollectOrbEnabled then break end
        for pass = 1, 5 do
            if not State.autoCollectOrbEnabled then break end
            for i = 1, 3 do teleport(spotPos); task.wait(0.15) end
            local orbs = getUncollectedOrbs()
            if #orbs == 0 then break end
            local collectedThisPass = false
            for _, orb in ipairs(orbs) do
                if not State.autoCollectOrbEnabled then break end
                local rootPos = orb.root.Position
                for i = 1, 3 do teleport(rootPos); task.wait(0.15) end
                local hrp = getHRP()
                if hrp and (hrp.Position - rootPos).Magnitude <= 20 then
                    for _ = 1, 4 do
                        pcall(function() fireproximityprompt(orb.prompt) end)
                        task.wait(0.1)
                    end
                    labelOrb(orb.root)
                    collectedThisPass = true
                    for i = 1, 3 do teleport(spotPos); task.wait(0.15) end
                end
                task.wait(0.1)
            end
            if not collectedThisPass and pass > 1 then break end
            task.wait(0.2)
        end
    end
    State.collectingOrbs = false
end

local function startTrialLoop()
    if State.trialLoop then return end
    State.inGamemodeTrial = true

    State.trialLoop = task.spawn(function()
        while State.autoJoinTrialEnabled do
            if not hasGamemodeTrial() then break end

            local num, kind = getCurrentWaveOrStage()
            if kind == "Stage" and num and num >= State.trialStopStage then
                Library:Notify({ Title = "Auto Trial", Description = "Stage " .. num .. " reached! Quitting.", Time = 3 })
                fireQuitGamemode()
                task.wait(1)
                break
            end

            local pos = getTrialSpotPos()
            if pos then teleport(pos) end
            task.wait(0.5)
        end

        if State.autoCollectOrbEnabled then
            collectAllOrbs()
        end
        State.inGamemodeTrial = false
        State.trialLoop = nil
    end)
end

local function startCreateRaidLoop()
    if State.raidLoop then return end

    State.raidLoop = task.spawn(function()
        while State.autoCreateRaidEnabled do

            if State.inGamemodeTrial or State.collectingOrbs then
                task.wait(0.5)
                continue
            end

            if isGamemodeGuiActive() and isGuiShowingWave() then
                local num, kind = getCurrentWaveOrStage()
                if kind == "Wave" and num and num >= State.raidStopWave then
                    Library:Notify({ Title = "Auto Raid", Description = "Wave " .. num .. " reached! Recreating raid.", Time = 3 })
                    fireQuitGamemode()
                    task.wait(2)
                    fireCreateRaid(State.raidMap)
                    task.wait(1)
                    fireStartRaid()
                    task.wait(2)
                else
                    local raidFolder = getMyRaidFolder()
                    local spotPos = getRaidSpotPos(raidFolder, "5")
                        or getRaidSpotPos(raidFolder, "4")
                        or getRaidSpotPos(raidFolder, "3")
                        or getRaidSpotPos(raidFolder, "2")
                        or getRaidSpotPos(raidFolder, "1")

                    if spotPos then
                        teleport(spotPos)
                    else
                        local spawnPos = getRaidSpawnPartPos(raidFolder)
                        if spawnPos then
                            for i = 1, 6 do
                                teleport(spawnPos)
                                task.wait(0.2)
                            end
                        end
                    end
                end
                task.wait(0.5)
                continue
            end

            if not isGamemodeGuiActive() then
                fireCreateRaid(State.raidMap)
                task.wait(1)
                fireStartRaid()
                task.wait(2)
            end

            task.wait(0.5)
        end

        State.raidLoop = nil
    end)
end

local function readPortalCount(rank)
    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return 0 end
    local inv = pg:FindFirstChild("CenterGUI") and pg.CenterGUI:FindFirstChild("Inventory")
    if not inv then return 0 end
    local wasEnabled = inv.Enabled
    inv.Enabled = true
    local scroll = inv:FindFirstChild("Main")
        and inv.Main:FindFirstChild("Content")
        and inv.Main.Content:FindFirstChild("Scroll")
    if not scroll then inv.Enabled = wasEnabled; return 0 end
    local node = PORTAL_RANK_MAP[rank]
    local frame = node and scroll:FindFirstChild(node)
    if not frame then inv.Enabled = wasEnabled; return 0 end
    local lbl2 = frame:FindFirstChild("Label2")
    local ctx = lbl2 and (lbl2.ContentText ~= "" and lbl2.ContentText or lbl2.Text) or ""
    local count = tonumber(ctx:match("x(%d+)")) or 0
    inv.Enabled = wasEnabled
    return count
end

local function getNextAvailableRank()
    for _, rank in ipairs(PORTAL_RANK_ORDER) do
        local selected = false
        for _, r in ipairs(State.portalRanks) do
            if r == rank then selected = true; break end
        end
        if selected and readPortalCount(rank) > 0 then
            return rank
        end
    end
    return nil
end

local function startCreatePortalLoop()
    if State.portalLoop then return end

    State.portalLoop = task.spawn(function()
        while State.autoCreatePortalEnabled do

            if State.inGamemodeTrial or State.collectingOrbs then
                task.wait(0.5)
                continue
            end

            if isGamemodeGuiActive() and isGuiShowingWave() then
                local num, kind = getCurrentWaveOrStage()
                if kind == "Wave" and num and num >= State.portalStopWave then
                    Library:Notify({ Title = "Auto Portal", Description = "Wave " .. num .. " reached! Recreating portal.", Time = 3 })
                    fireQuitGamemode()
                    task.wait(2)
                    local rank = getNextAvailableRank()
                    if rank then
                        fireCreatePortal(rank)
                        task.wait(1)
                        fireStartPortal()
                        task.wait(2)
                    else
                        Library:Notify({ Title = "Auto Portal", Description = "No portals left! Stopping.", Time = 4 })
                        State.autoCreatePortalEnabled = false
                        Toggles.AutoCreatePortalToggle:SetValue(false)
                        break
                    end
                else
                    local portalFolder = getMyPortalFolder()
                    local spotPos = getPortalSpotPos(portalFolder, "5")
                        or getPortalSpotPos(portalFolder, "4")
                        or getPortalSpotPos(portalFolder, "3")
                        or getPortalSpotPos(portalFolder, "2")
                        or getPortalSpotPos(portalFolder, "1")

                    if spotPos then
                        teleport(spotPos)
                    else
                        local spawnPos = getPortalSpawnPartPos(portalFolder)
                        if spawnPos then
                            for i = 1, 6 do
                                teleport(spawnPos)
                                task.wait(0.2)
                            end
                        end
                    end
                end
                task.wait(0.5)
                continue
            end

            if not isGamemodeGuiActive() then
                local rank = getNextAvailableRank()
                if rank then
                    fireCreatePortal(rank)
                    task.wait(1)
                    fireStartPortal()
                    task.wait(2)
                else
                    Library:Notify({ Title = "Auto Portal", Description = "No portals left! Stopping.", Time = 4 })
                    State.autoCreatePortalEnabled = false
                    Toggles.AutoCreatePortalToggle:SetValue(false)
                    break
                end
            end

            task.wait(0.5)
        end

        State.portalLoop = nil
    end)
end

local function startFarmLoop()
    if State.farmLoop then return end

    State.farmLoop = task.spawn(function()
        while State.autofarmEnabled do
            clickConfirmIfInviteOpen()

            if State.autoCreateRaidEnabled and isGamemodeGuiActive() then
                task.wait(0.5)
                continue
            end

            if State.autoCreatePortalEnabled and isGamemodeGuiActive() then
                task.wait(0.5)
                continue
            end

            if State.inGamemodeTrial or State.collectingOrbs then
                task.wait(0.3)
                continue
            end

            local selected = State.selectedSpawns
            if #selected == 0 then task.wait(0.5); continue end

            local spawnName = selected[State.currentSpawnIdx]
            local spawnPos  = SPAWN_MAP[spawnName]

            if not spawnPos then
                State.currentSpawnIdx = (State.currentSpawnIdx % #selected) + 1
                task.wait(0.1)
                continue
            end

            local t0 = tick()
            while tick() - t0 < 2 do
                if not State.autofarmEnabled then break end
                teleport(spawnPos)
                task.wait(0.1)
            end

            while State.autofarmEnabled do
                if State.autoCreateRaidEnabled and isGamemodeGuiActive() then break end
                if State.autoCreatePortalEnabled and isGamemodeGuiActive() then break end
                if hasGamemodeTrial() then break end
                if State.collectingOrbs then break end
                clickConfirmIfInviteOpen()
                teleport(spawnPos)
                if not hasLiveEnemiesInRange() then break end
                fireAttack(getInRangeIDs())
                task.wait(0.1)
            end

            State.currentSpawnIdx = (State.currentSpawnIdx % #selected) + 1
        end

        State.farmLoop = nil
    end)
end

local function stopFarmLoop()
    State.autofarmEnabled = false
    State.farmLoop = nil
end

local function setupGamemodeWatcher()
    if State.gamemodeWatcher then
        State.gamemodeWatcher:Disconnect()
        State.gamemodeWatcher = nil
    end
    if not State.autoJoinTrialEnabled then return end

    local ok, gm = pcall(function()
        return workspace:WaitForChild("_MAP", 5):WaitForChild("Gamemode", 5)
    end)
    if not ok or not gm then return end

    local function onTrialFound()
        task.delay(0.5, function()
            if not State.autoJoinTrialEnabled then return end
            fireEnterTrial(State.trialMap)
            startTrialLoop()
        end)
    end

    for _, child in ipairs(gm:GetChildren()) do
        if child.Name:match("^Time Trial_") then onTrialFound(); break end
    end

    State.gamemodeWatcher = gm.ChildAdded:Connect(function(child)
        if not State.autoJoinTrialEnabled then return end
        if child.Name:match("^Time Trial_") then onTrialFound() end
    end)
end

local function setupInviteWatcher()
    if State.inviteWatcherConn then
        State.inviteWatcherConn:Disconnect()
        State.inviteWatcherConn = nil
    end
    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return end
    local function watchInvite(inv)
        if not inv then return end
        inv:GetPropertyChangedSignal("Enabled"):Connect(function()
            if not inv.Enabled then return end
            if not State.autoJoinTrialEnabled then return end
            task.spawn(function()
                task.wait(5)
                if not inv.Enabled then return end
                if not State.autoJoinTrialEnabled then return end
                if isGamemodeGuiActive() and isGuiShowingWave() then
                    fireQuitGamemode()
                    task.wait(1.5)
                end
                fireEnterTrial(State.trialMap)
            end)
        end)
    end
    local inv = pg:FindFirstChild("Invite")
    if inv then
        watchInvite(inv)
    else
        State.inviteWatcherConn = pg.ChildAdded:Connect(function(child)
            if child.Name == "Invite" then watchInvite(child) end
        end)
    end
end

local function handleGamemodeChild(child)
    if State.collectingOrbs then return end

    local raidID = child.Name:match("^Raid_(%d+)$")
    if raidID and State.autoJoinRaidEnabled then
        task.delay(0.5, function()
            if not State.autoJoinRaidEnabled then return end
            fireJoin("Raid", tonumber(raidID))
        end)
        return
    end

    local portalID = child.Name:match("^Portal_(%d+)$")
    if portalID and State.autoJoinPortalEnabled then
        task.delay(0.5, function()
            if not State.autoJoinPortalEnabled then return end
            fireJoin("Portal", tonumber(portalID))
        end)
    end
end

local function setupRaidPortalWatcher()
    if State.raidPortalWatcher then
        State.raidPortalWatcher:Disconnect()
        State.raidPortalWatcher = nil
    end
    if not State.autoJoinRaidEnabled and not State.autoJoinPortalEnabled then return end

    local mapFolder = workspace:FindFirstChild("_MAP")
    if not mapFolder then return end
    local ok, gm = pcall(function() return mapFolder:WaitForChild("Gamemode", 5) end)
    if not ok or not gm then return end

    for _, child in ipairs(gm:GetChildren()) do handleGamemodeChild(child) end

    State.raidPortalWatcher = gm.ChildAdded:Connect(function(child)
        handleGamemodeChild(child)
    end)
end

local function getAvailablePortalNames()
    local gm = workspace:FindFirstChild("_MAP") and workspace._MAP:FindFirstChild("Gamemode")
    if not gm then return {} end
    local names = {}
    for _, child in ipairs(gm:GetChildren()) do
        local id = child.Name:match("^Portal_(%d+)$")
        if id then table.insert(names, "Portal " .. id) end
    end
    return names
end

local function startForcePortalLoop()
    if State.forcePortalLoop then return end
    State.forcePortalLoop = task.spawn(function()
        while State.autoForcePortalEnabled do
            if State.forcePortalID then fireJoin("Portal", State.forcePortalID) end
            task.wait(10)
        end
        State.forcePortalLoop = nil
    end)
end
