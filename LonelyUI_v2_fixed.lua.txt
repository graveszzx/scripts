local PROFILES_FILE = "LonelyUI_Profiles.json"

local function diskRead()
    local ok, data = pcall(readfile, PROFILES_FILE)
    if ok and data and data ~= "" then
        local ok2, t = pcall(HttpService.JSONDecode, HttpService, data)
        if ok2 then return t end
    end
    return { profiles = {}, active = nil }
end

local function diskWrite(t)
    local ok, str = pcall(HttpService.JSONEncode, HttpService, t)
    if ok then pcall(writefile, PROFILES_FILE, str) end
end

local function listProfiles()
    return diskRead().profiles
end

local function saveProfile(name, data)
    local store = diskRead()
    store.profiles[name] = data
    store.active = name
    diskWrite(store)
end

local function loadProfile(name)
    local store = diskRead()
    return store.profiles[name]
end

local function deleteProfile(name)
    local store = diskRead()
    store.profiles[name] = nil
    if store.active == name then store.active = nil end
    diskWrite(store)
end

local function isMobile()
    return UserInputService.TouchEnabled and not UserInputService.MouseEnabled
end

local _registry = {}
local function registerElement(id, getfn, setfn, etype)
    _registry[id] = { get = getfn, set = setfn, type = etype }
end

local function collectConfig()
    local t = {}
    for id, el in pairs(_registry) do
        local ok, v = pcall(el.get)
        if ok then
            t[id] = { type = el.type, value = v }
        end
    end
    return t
end

local function applyConfig(t, execute)
    if not t then return end
    for id, data in pairs(t) do
        local el = _registry[id]
        if el and el.set then
            pcall(function()
                if data.type == "segment" then
                    el.set(nil, data.value)
                elseif data.type == "toggle" or data.type == "selectable" then
                    el.set(nil, data.value)
                    if execute then el.get() end
                else
                    el.set(nil, data.value)
                end
            end)
        end
    end
end

            local function wireKeybind(kb, kbs, onFire)
                local boundKey = nil
                local binding  = false

                local function refresh()
                    if binding then
                        kb.Text = "..."; kb.TextColor3 = Theme.Accent; kbs.Color = Theme.Accent
                    elseif boundKey then
                        kb.Text = boundKey; kb.TextColor3 = Theme.Accent; kbs.Color = Theme.Accent
                    else
                        kb.Text = "bind"; kb.TextColor3 = Color3.fromRGB(90,90,100); kbs.Color = Color3.fromRGB(60,60,70)
                    end
                end

                kb.MouseButton1Click:Connect(function()
                    if binding then binding = false; refresh(); return end
                    binding = true; refresh()
                end)

                UserInputService.InputBegan:Connect(function(input, gp)
                    if gp then return end
                    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
                    if binding then
                        if input.KeyCode == Enum.KeyCode.Escape then binding = false; refresh(); return end
                        boundKey = input.KeyCode.Name; binding = false; refresh()
                        return
                    end
                    if boundKey and input.KeyCode.Name == boundKey then onFire() end
                end)

                return {
                    GetKey = function(_) return boundKey end,
                    SetKey = function(_, k) boundKey = k; refresh() end,
                }
            end

        Callback = function()
            local name = profileNameBox:Get()
            if name == "" then return end
            local data = collectConfig()
            saveProfile(name, data)
        end,
    })

    local profileListCtrl = nil
    local function rebuildProfileList()
        local profiles = listProfiles()
        local names = {}
        for k in pairs(profiles) do
            table.insert(names, k)
        end
        table.sort(names)
        if #names == 0 then
            names = { "(nenhum)" }
        end
        if profileListCtrl then
            profileListCtrl:Set(nil, names[1])
        end
        return names
    end

    local initialNames = rebuildProfileList()

    local ddRow = Instance.new("Frame")
    ddRow.ZIndex = 6; ddRow.LayoutOrder = 999
    ddRow.Size = UDim2.new(1,-2,0,40)
    ddRow.BackgroundColor3 = Color3.fromRGB(255,255,255)
    ddRow.BackgroundTransparency = 0.82; ddRow.BorderSizePixel = 0
    ddRow.Parent = configTab._page
    local ddRowC = Instance.new("UICorner"); ddRowC.CornerRadius = UDim.new(0,10); ddRowC.Parent = ddRow
    local ddRowS = Instance.new("UIStroke"); ddRowS.Color = LonelyUI.Theme.Border; ddRowS.Transparency = 0.6; ddRowS.Parent = ddRow

    local ddLbl = Instance.new("TextLabel")
    ddLbl.ZIndex = 8; ddLbl.Position = UDim2.new(0,11,0,0)
    ddLbl.Size = UDim2.new(0.58,0,1,0); ddLbl.BackgroundTransparency = 1
    ddLbl.Text = "Perfil ativo"; ddLbl.TextColor3 = LonelyUI.Theme.TextPrimary; ddLbl.TextSize = 11
    ddLbl.Font = Enum.Font.GothamBold; ddLbl.TextXAlignment = Enum.TextXAlignment.Left; ddLbl.Parent = ddRow

    local selectedProfile = initialNames[1]
    local ddValLbl = Instance.new("TextLabel"); ddValLbl.ZIndex = 5
    ddValLbl.Position = UDim2.new(1,-100,0,0); ddValLbl.Size = UDim2.new(0,80,1,0)
    ddValLbl.BackgroundTransparency = 1; ddValLbl.Text = selectedProfile
    ddValLbl.TextColor3 = LonelyUI.Theme.Accent; ddValLbl.TextSize = 10
    ddValLbl.Font = Enum.Font.GothamBold; ddValLbl.TextXAlignment = Enum.TextXAlignment.Right; ddValLbl.Parent = ddRow

    local ddArrow = Instance.new("TextLabel"); ddArrow.ZIndex = 5
    ddArrow.Position = UDim2.new(1,-18,0,0); ddArrow.Size = UDim2.new(0,14,1,0)
    ddArrow.BackgroundTransparency = 1; ddArrow.Text = "▾"
    ddArrow.TextColor3 = LonelyUI.Theme.Accent; ddArrow.TextSize = 12
    ddArrow.Font = Enum.Font.GothamBold; ddArrow.Parent = ddRow

    local ddClickLayer = Instance.new("TextButton"); ddClickLayer.ZIndex = 7
    ddClickLayer.Size = UDim2.new(1,0,1,0); ddClickLayer.BackgroundTransparency = 1
    ddClickLayer.Text = ""; ddClickLayer.Parent = ddRow

    local ddPanel = Instance.new("Frame")
    ddPanel.Name = "DDPanel_Profiles"; ddPanel.ZIndex = 201; ddPanel.Visible = false
    ddPanel.Size = UDim2.new(0,160,0,100)
    ddPanel.BackgroundColor3 = Color3.fromRGB(14,14,16)
    ddPanel.BackgroundTransparency = 0.05; ddPanel.BorderSizePixel = 0
    ddPanel.Parent = _dropdownOverlay
    local ddPanelC = Instance.new("UICorner"); ddPanelC.CornerRadius = UDim.new(0,10); ddPanelC.Parent = ddPanel
    local ddPanelS = Instance.new("UIStroke"); ddPanelS.Color = LonelyUI.Theme.Accent; ddPanelS.Transparency = 0.4; ddPanelS.Parent = ddPanel
    local ddPanelUL = Instance.new("UIListLayout"); ddPanelUL.Padding = UDim.new(0,2); ddPanelUL.Parent = ddPanel
    local ddPanelPad = Instance.new("UIPadding")
    ddPanelPad.PaddingTop = UDim.new(0,4); ddPanelPad.PaddingBottom = UDim.new(0,4)
    ddPanelPad.PaddingLeft = UDim.new(0,4); ddPanelPad.PaddingRight = UDim.new(0,4); ddPanelPad.Parent = ddPanel

    local ddOpen = false
    local _ddJustOpened = false

    local function rebuildDDPanel(names)
        for _, ch in ipairs(ddPanel:GetChildren()) do
            if ch:IsA("TextButton") then ch:Destroy() end
        end
        local panelH = #names * 28 + 8
        ddPanel.Size = UDim2.new(0,160,0,panelH)
        for _, name in ipairs(names) do
            local optBtn = Instance.new("TextButton"); optBtn.ZIndex = 202
            optBtn.Size = UDim2.new(1,0,0,24)
            optBtn.BackgroundColor3 = Color3.fromRGB(22,22,26)
            optBtn.BackgroundTransparency = 0.3; optBtn.BorderSizePixel = 0
            optBtn.Text = name
            optBtn.TextColor3 = name == selectedProfile and LonelyUI.Theme.Accent or LonelyUI.Theme.TextSecondary
            optBtn.TextSize = 10; optBtn.Font = Enum.Font.GothamBold; optBtn.Parent = ddPanel
            local oc = Instance.new("UICorner"); oc.CornerRadius = UDim.new(0,6); oc.Parent = optBtn
            optBtn.MouseButton1Click:Connect(function()
                selectedProfile = name; ddValLbl.Text = name
                ddPanel.Visible = false; ddOpen = false
                tw(ddArrow, 0.1, {Rotation = 0})
            end)
        end
    end

    local function positionDDPanel()
        local abs = ddRow.AbsolutePosition
        local sz  = ddRow.AbsoluteSize
        local sy  = ScreenGui.AbsoluteSize.Y
        local py  = abs.Y + sz.Y + 4
        local panelH = ddPanel.AbsoluteSize.Y
        if py + panelH > sy then py = abs.Y - panelH - 4 end
        ddPanel.Position = UDim2.new(0, abs.X, 0, py)
        ddPanel.Size = UDim2.new(0, sz.X, 0, ddPanel.AbsoluteSize.Y)
    end

    rebuildDDPanel(initialNames)

    ddClickLayer.MouseButton1Click:Connect(function()
        local names = rebuildProfileList()
        rebuildDDPanel(names)
        ddOpen = not ddOpen
        _ddJustOpened = ddOpen
        if ddOpen then positionDDPanel() end
        ddPanel.Visible = ddOpen
        tw(ddArrow, 0.15, {Rotation = ddOpen and 180 or 0})
        if ddOpen then
            task.delay(0.05, function() _ddJustOpened = false end)
        end
    end)

    UserInputService.InputBegan:Connect(function(i)
        if not ddOpen then return end
        if _ddJustOpened then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            task.wait()
            if ddPanel.Visible then
                ddPanel.Visible = false; ddOpen = false
                tw(ddArrow, 0.1, {Rotation = 0})
            end
        end
    end)

    local actionsSec = configTab:AddSection("Ações")

    actionsSec:AddButton("Carregar perfil", {
        Label    = "Carregar perfil",
        Callback = function()
            if selectedProfile == "(nenhum)" then return end
            local data = loadProfile(selectedProfile)
            applyConfig(data, false)
        end,
    })

    actionsSec:AddButton("Carregar e executar", {
        Label    = "Carregar e executar",
        Callback = function()
            if selectedProfile == "(nenhum)" then return end
            local data = loadProfile(selectedProfile)
            applyConfig(data, true)
        end,
    })

    actionsSec:AddButton("Deletar perfil", {
        Label    = "Deletar perfil",
        Callback = function()
            if selectedProfile == "(nenhum)" then return end
            deleteProfile(selectedProfile)
            local names = rebuildProfileList()
            rebuildDDPanel(names)
            selectedProfile = names[1]
            ddValLbl.Text = selectedProfile
        end,
    })

    function Window:SetThemeColor(color)
        Theme.Accent  = color
        Theme.ToggleOn = color
        Theme.Border  = color
    end

    function Window:Destroy()
        ScreenGui:Destroy()
    end

    function Window:SaveProfile(name)
        local data = collectConfig()
        saveProfile(name, data)
        return data
    end

    function Window:LoadProfile(name, execute)
        local data = loadProfile(name)
        applyConfig(data, execute)
    end

    function Window:ListProfiles()
        return listProfiles()
    end

    function Window:SetMinimised(v)
        setMinimised(v)
    end

    return Window
end

return LonelyUI
